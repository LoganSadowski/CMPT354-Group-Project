CREATE TABLE Technician (
	technicianID INT NOT NULL,
	firstName VARCHAR(50),
	lastName VARCHAR(50),
	email VARCHAR(50),
	phone VARCHAR(15),
	depID INT NOT NULL,
	PRIMARY KEY (technicianID),
	FOREIGN KEY (depID) REFERENCES Department (departmentID),
	CONSTRAINT uniqueDep UNIQUE (technicianID, depID)
);

CREATE TABLE Department (
	departmentID INT NOT NULL,
	name VARCHAR(50),
	managerID INT NOT NULL,
	PRIMARY KEY (departmentID),
	FOREIGN KEY (managerID, departmentID) REFERENCES Technician (technicianID, depID)
);

CREATE TABLE Client (
	clientID INT NOT NULL,
	email VARCHAR(50),
	PRIMARY KEY (clientID)
);

CREATE TABLE Company (
	cID INT NOT NULL,
	name VARCHAR(50),
	PRIMARY KEY (cID),
	FOREIGN KEY (cID) REFERENCES Client (clientID)
);

CREATE TABLE Individual (
	cID INT NOT NULL,
	firstName VARCHAR(50),
	lastName VARCHAR(50),
	PRIMARY KEY (cID),
	FOREIGN KEY (cID) REFERENCES Client (clientID)
);

CREATE TABLE StorageLocation (
	storageID INT NOT NULL,
	address VARCHAR(50),
	capacity INT,
	PRIMARY KEY (storageID)
);

CREATE TABLE Sample (
	sampleID INT NOT NULL,
	description VARCHAR(300),
	dateReceived VARCHAR(50),
	cID INT NOT NULL,
	sID INT,
	PRIMARY KEY (sampleID),
	FOREIGN KEY (cID) REFERENCES Client (clientID),
	FOREIGN KEY (sID) REFERENCES StorageLocation (storageID)
);

CREATE TABLE SubSample (
	subSampleID CHAR NOT NULL,
	sampleID INT NOT NULL,
	parentSubSampleID CHAR,
	PRIMARY KEY (subSampleID, sampleID),
	FOREIGN KEY (sampleID) REFERENCES Sample (sampleID)
);

CREATE TRIGGER trg_subsample_parent_check_insert
BEFORE INSERT ON SubSample
FOR EACH ROW
WHEN NEW.parentSubSampleID IS NOT NULL
BEGIN
	SELECT CASE
		WHEN NEW.parentSubSampleID = NEW.subSampleID
			THEN RAISE(ABORT, 'SubSample cannot be its own parent')
		WHEN NOT EXISTS (
			SELECT 1
			FROM SubSample s
			WHERE s.subSampleID = NEW.parentSubSampleID
			  AND s.sampleID = NEW.sampleID
		)
			THEN RAISE(ABORT, 'Parent SubSample must exist in same Sample')
	END;
END;

CREATE TRIGGER trg_subsample_parent_check_update
BEFORE UPDATE OF parentSubSampleID, subSampleID, sampleID ON SubSample
FOR EACH ROW
WHEN NEW.parentSubSampleID IS NOT NULL
BEGIN
	SELECT CASE
		WHEN NEW.parentSubSampleID = NEW.subSampleID
			THEN RAISE(ABORT, 'SubSample cannot be its own parent')
		WHEN NOT EXISTS (
			SELECT 1
			FROM SubSample s
			WHERE s.subSampleID = NEW.parentSubSampleID
			  AND s.sampleID = NEW.sampleID
		)
			THEN RAISE(ABORT, 'Parent SubSample must exist in same Sample')
	END;
END;

CREATE TABLE TestRun (
	testRunID INT NOT NULL,
	sampleID INT NOT NULL,
	technicianID INT NOT NULL,
	testType VARCHAR(50) NOT NULL,
	performedAt DATETIME,
	instrument VARCHAR(50),
	status VARCHAR(20) NOT NULL,
	notes VARCHAR(300),
	PRIMARY KEY (testRunID),
	FOREIGN KEY (sampleID) REFERENCES Sample (sampleID),
	FOREIGN KEY (technicianID) REFERENCES Technician (technicianID)
);

CREATE TABLE Analyte (
	analyteID INT NOT NULL,
	name VARCHAR(100) NOT NULL,
	defaultUnit VARCHAR(20),
	dataType VARCHAR(20) NOT NULL,
	PRIMARY KEY (analyteID)
);

CREATE TABLE TestMeasurement (
	testRunID INT NOT NULL,
	analyteID INT NOT NULL,
	valueNumeric DECIMAL(12,4),
	valueText VARCHAR(200),
	PRIMARY KEY (testRunID, analyteID),
	FOREIGN KEY (testRunID) REFERENCES TestRun (testRunID),
	FOREIGN KEY (analyteID) REFERENCES Analyte (analyteID)
);
